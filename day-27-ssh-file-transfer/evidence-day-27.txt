SSH KEYS
--------
ls -la ~/.ssh
ssh-keygen -t ed25519 -f ~/.ssh/day27_ed25519 -C "day27-linux-challenge"
ssh-keygen -lf ~/.ssh/day27_ed25519.pub

FILE TRANSFER TEST
------------------
printf 'Day 27 secure file transfer practice\n' > transfer-test.txt
mkdir -p rsync-destination
rsync -av transfer-test.txt rsync-destination/
cat rsync-destination/transfer-test.txt

-----------------

chmod 700 ~/.ssh
chmod 600 ~/.ssh/config
